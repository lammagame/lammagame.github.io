rules_version = '2';
// قواعد «فقرات الجمهور»: النشر بدون حساب (دخول مجهول)، القراءة للكل،
// البلاغ مرة وحدة لكل جهاز، 3 بلاغات تخفي الفقرة، والمشرف يقدر يخفي ويحذف ويحظر.
service cloud.firestore {
  match /databases/{db}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() { return signedIn() && request.auth.token.email == "lammagame.app@gmail.com" && request.auth.token.email_verified == true; }
    function banned() { return exists(/databases/$(db)/documents/bans/$(request.auth.uid)); }
    function onlyChanged(keys) { return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys); }

    match /crowd/{id} {
      allow read: if resource.data.hidden == false || isAdmin() || (signedIn() && resource.data.uid == request.auth.uid);
      allow create: if signedIn() && !banned()
        && request.resource.data.keys().hasOnly(["name","author","uid","q","n","plays","reports","hidden","created"])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40
        && request.resource.data.author is string && request.resource.data.author.size() <= 24
        && request.resource.data.q is list && request.resource.data.q.size() >= 6 && request.resource.data.q.size() <= 300
        && request.resource.data.n == request.resource.data.q.size()
        && request.resource.data.plays == 0 && request.resource.data.reports == 0 && request.resource.data.hidden == false
        && request.resource.data.created == request.time;
      // عدّاد اللعب: +1 فقط
      allow update: if signedIn() && onlyChanged(["plays"]) && request.resource.data.plays == resource.data.plays + 1;
      // البلاغ: +1 ويا مستند بلاغ جديد بنفس العملية، والإخفاء تلقائي عند 3
      allow update: if signedIn() && onlyChanged(["reports","hidden"])
        && request.resource.data.reports == resource.data.reports + 1
        && request.resource.data.hidden == (request.resource.data.reports >= 3 || resource.data.hidden)
        && !exists(/databases/$(db)/documents/reports/$(id + "_" + request.auth.uid))
        && existsAfter(/databases/$(db)/documents/reports/$(id + "_" + request.auth.uid));
      allow update: if isAdmin();
      allow delete: if isAdmin() || (signedIn() && resource.data.uid == request.auth.uid);
    }
    // صور الفقرات المنشورة (مصغّرة، أقل من ~700KB)
    match /media/{mid} {
      allow read: if true;
      allow create: if signedIn() && !banned() && request.resource.data.keys().hasOnly(["cat","uid","data"])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.data is string && request.resource.data.data.size() < 700000
        && request.resource.data.data.matches("^data:image/jpeg;base64,.*");
      allow delete: if isAdmin() || (signedIn() && resource.data.uid == request.auth.uid);
    }
    match /reports/{rid} {
      allow create: if signedIn() && rid == request.resource.data.cat + "_" + request.auth.uid && request.resource.data.uid == request.auth.uid;
      allow read, delete: if isAdmin();
    }
    // إحصائيات مجهولة: عدّادات فقط (كم مرة انلعبت الفقرة، ومن أي منطقة زمنية). القراءة للمشرف بس
    match /stats/{d} {
      // cats/geo/time = المجموع الكلي. d_YYYY-MM-DD = عدّادات كل يوم
      function statDoc() { return d in ["cats", "geo", "time"] || d.matches("^d_[0-9]{4}-[0-9]{2}-[0-9]{2}$"); }
      allow read: if isAdmin();
      allow create: if signedIn() && statDoc() && request.resource.data.keys().size() <= 8;
      allow update: if signedIn() && statDoc() && request.resource.data.diff(resource.data).affectedKeys().size() <= 8;
    }
    match /bans/{uid} {
      allow read, write: if isAdmin();
    }
  }
}
